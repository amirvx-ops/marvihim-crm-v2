SELECT 
  insurance_company AS חברה,
  COUNT(*) AS לידים,
  COUNT(CASE WHEN closing_amount > 0 THEN 1 END) AS סגירות,
  ROUND(SUM(COALESCE(closing_amount, 0))) AS סכום_סגירות,
  CASE WHEN COUNT(*) > 0
    THEN ROUND(COUNT(CASE WHEN closing_amount > 0 THEN 1 END)::numeric / COUNT(*)::numeric * 100, 1)
    ELSE 0 END AS אחוז_סגירה
FROM leads
WHERE 
  insurance_company IS NOT NULL AND insurance_company != ''
  AND ('{{Select_Month.selectedOptionValue}}' = '' 
       OR TO_CHAR(user_updated_at, 'MM/YYYY') = '{{Select_Month.selectedOptionValue}}')
GROUP BY insurance_company
ORDER BY סגירות DESC;
