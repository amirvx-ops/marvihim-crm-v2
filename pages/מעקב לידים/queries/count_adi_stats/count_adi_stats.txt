SELECT
    COUNT(*) AS total,
    COUNT(CASE WHEN חברת_ביטוח = 'הפניקס' THEN 1 END) AS פניקס,
    COUNT(CASE WHEN חברת_ביטוח = 'הראל' THEN 1 END) AS הראל,
    COUNT(CASE WHEN חברת_ביטוח = 'כלל' THEN 1 END) AS כלל,
    COALESCE(SUM(מינוי_סוכן), 0) AS סהכ_מינוי,
    COALESCE(SUM(סכום_סגירה), 0) AS סהכ_סגירה
FROM adi_leads_view
WHERE touched_by_adi = TRUE
AND ('{{Select_Month.selectedOptionValue}}' = '' OR TO_CHAR(תאריך_עדכון, 'MM/YYYY') = '{{Select_Month.selectedOptionValue}}')
