SELECT * FROM adi_leads_view
WHERE (
    '{{Input_Search.text}}' != ''
    AND (
        "שם_לקוח" ILIKE '%' || '{{Input_Search.text}}' || '%'
        OR CAST("תעודת_זהות" AS TEXT) ILIKE '%' || '{{Input_Search.text}}' || '%'
    )
)
OR (
    '{{Input_Search.text}}' = ''
    AND (
        ('{{Select_Agent.selectedOptionValue}}' = 'עדי' AND "שיוך" = 'עדי')
        OR ('{{Select_Agent.selectedOptionValue}}' = 'יולי' AND "שיוך" = 'יולי')
        OR ('{{Select_Agent.selectedOptionValue}}' = 'כולם' AND "שיוך" IS NOT NULL)
        OR ('{{Select_Agent.selectedOptionValue}}' = 'ללא שיוך' AND "שיוך" IS NULL)
        OR ('{{Select_Agent.selectedOptionValue}}' = '' AND "שיוך" = 'עדי')
    )
    AND ('{{Select_Month.selectedOptionValue}}' = '' OR TO_CHAR("תאריך_עדכון", 'MM/YYYY') = '{{Select_Month.selectedOptionValue}}')
    AND ('{{status.selectedOptionValue}}' = '' OR "סטטוס" = '{{status.selectedOptionValue}}')
    AND ('{{insurance_company.selectedOptionValue}}' = '' OR "חברת_ביטוח" = '{{insurance_company.selectedOptionValue}}')
    AND ('{{Select_Partner.selectedOptionValue}}' = '' OR "מקור_ליד" = '{{Select_Partner.selectedOptionValue}}')
)
ORDER BY "תאריך_עדכון" DESC NULLS LAST
LIMIT 500
