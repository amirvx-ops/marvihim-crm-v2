SELECT TO_CHAR(DATE_TRUNC('month', created_at), 'MM/YYYY') AS x, COUNT(CASE WHEN closing_amount > 0 THEN 1 END)::int AS y FROM leads WHERE created_at >= CURRENT_DATE - INTERVAL '12 months' AND (('{{Select_Agent.selectedOptionValue}}' = 'עדי' AND touched_by_adi = TRUE) OR ('{{Select_Agent.selectedOptionValue}}' = 'יולי' AND touched_by_yuli = TRUE) OR ('{{Select_Agent.selectedOptionValue}}' = 'כולם' AND (touched_by_adi = TRUE OR touched_by_yuli = TRUE)) OR ('{{Select_Agent.selectedOptionValue}}' = '' AND (touched_by_adi = TRUE OR touched_by_yuli = TRUE))) GROUP BY DATE_TRUNC('month', created_at) ORDER BY DATE_TRUNC('month', created_at)
