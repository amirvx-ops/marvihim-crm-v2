SELECT
  agent AS מטפל,
  COUNT(*) AS לידים_שעברו,
  COUNT(CASE WHEN deal_closed IS NOT NULL AND deal_closed != '' THEN 1 END) AS עסקאות,
  CASE WHEN COUNT(*) > 0
    THEN ROUND(COUNT(CASE WHEN deal_closed IS NOT NULL AND deal_closed != '' THEN 1 END)::numeric / COUNT(*)::numeric * 100, 1)
    ELSE 0 END AS אחוז_סגירה
FROM lidor_leads_view
WHERE agent IS NOT NULL
  AND ('{{Select_Month_Lidor.selectedOptionValue}}' = ''
       OR TO_CHAR(update_date, 'MM/YYYY') = '{{Select_Month_Lidor.selectedOptionValue}}')
GROUP BY agent
ORDER BY לידים_שעברו DESC;
