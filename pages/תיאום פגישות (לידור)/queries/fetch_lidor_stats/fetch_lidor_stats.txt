SELECT
  COUNT(*) AS total,
  COUNT(CASE WHEN meeting_date IS NOT NULL THEN 1 END) AS meetings,
  COUNT(CASE WHEN lead_passed = TRUE THEN 1 END) AS passed,
  COUNT(CASE WHEN deal_closed = 'כן' THEN 1 END) AS deals,
  CASE WHEN COUNT(*) > 0
    THEN ROUND(COUNT(CASE WHEN lead_passed = TRUE THEN 1 END)::numeric / COUNT(*)::numeric * 100, 1)
    ELSE 0 END AS pass_rate,
  CASE WHEN COUNT(*) > 0
    THEN ROUND(COUNT(CASE WHEN deal_closed = 'כן' THEN 1 END)::numeric / COUNT(*)::numeric * 100, 1)
    ELSE 0 END AS close_rate,
  ROUND(AVG(CASE WHEN meeting_date IS NOT NULL
    THEN EXTRACT(DAY FROM meeting_date - created_at) END), 1) AS avg_days_to_meeting
FROM lidor_leads_view
WHERE touched_by_lidor = TRUE
AND ('{{Select_Month_Lidor.selectedOptionValue}}' = ''
     OR TO_CHAR(update_date, 'MM/YYYY') = '{{Select_Month_Lidor.selectedOptionValue}}')
AND ('{{Select_Source.selectedOptionValue}}' = ''
     OR source = '{{Select_Source.selectedOptionValue}}')
AND ('{{Select_Agent.selectedOptionValue}}' = ''
     OR agent = '{{Select_Agent.selectedOptionValue}}');
