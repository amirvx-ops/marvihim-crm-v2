SELECT
  COUNT(*) AS total,
  COUNT(CASE WHEN meeting_date IS NOT NULL THEN 1 END) AS meetings,
  COUNT(CASE WHEN lead_passed = TRUE THEN 1 END) AS passed,
  COUNT(CASE WHEN deal_closed IS NOT NULL AND deal_closed != '' THEN 1 END) AS deals
FROM lidor_leads_view
WHERE touched_by_lidor = TRUE
AND ('{{Select_Month_Lidor.selectedOptionValue}}' = ''
     OR TO_CHAR(update_date, 'MM/YYYY') = '{{Select_Month_Lidor.selectedOptionValue}}');
