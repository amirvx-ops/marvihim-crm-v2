SELECT
  COALESCE(source, 'לא ידוע') AS מקור,
  COUNT(*) AS לידים,
  COUNT(CASE WHEN lead_passed = TRUE THEN 1 END) AS עברו,
  COUNT(CASE WHEN deal_closed = 'כן' THEN 1 END) AS סגרו,
  CASE WHEN COUNT(*) > 0
    THEN ROUND(COUNT(CASE WHEN lead_passed = TRUE THEN 1 END)::numeric / COUNT(*)::numeric * 100, 1)
    ELSE 0 END AS אחוז_העברה,
  CASE WHEN COUNT(*) > 0
    THEN ROUND(COUNT(CASE WHEN deal_closed = 'כן' THEN 1 END)::numeric / COUNT(*)::numeric * 100, 1)
    ELSE 0 END AS אחוז_סגירה
FROM lidor_leads_view
WHERE touched_by_lidor = TRUE
AND ('{{Select_Month_Lidor.selectedOptionValue}}' = ''
     OR TO_CHAR(update_date, 'MM/YYYY') = '{{Select_Month_Lidor.selectedOptionValue}}')
GROUP BY source
ORDER BY לידים DESC;
