SELECT
  מקור,
  COUNT(*) AS לידים_שעברו,
  COUNT(CASE WHEN נסגרה_עסקה IS NOT NULL AND נסגרה_עסקה != '' THEN 1 END) AS עסקאות,
  CASE WHEN COUNT(*) > 0
    THEN ROUND(COUNT(CASE WHEN נסגרה_עסקה IS NOT NULL AND נסגרה_עסקה != '' THEN 1 END)::numeric / COUNT(*)::numeric * 100, 1)
    ELSE 0 END AS אחוז_סגירה
FROM lidor_leads_view
WHERE ('{{Select_Month_Lidor.selectedOptionValue}}' = ''
       OR TO_CHAR(תאריך_עדכון, 'MM/YYYY') = '{{Select_Month_Lidor.selectedOptionValue}}')
GROUP BY מקור
ORDER BY לידים_שעברו DESC
