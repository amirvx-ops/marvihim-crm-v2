SELECT * FROM lidor_leads_view
WHERE (
    '{{Input_Search_Lidor.text}}' != ''
    AND (שם_לקוח ILIKE '%' || '{{Input_Search_Lidor.text}}' || '%'
         OR תעודת_זהות ILIKE '%' || '{{Input_Search_Lidor.text}}' || '%')
)
OR (
    '{{Input_Search_Lidor.text}}' = ''
    AND touched_by_lidor = TRUE
    AND ('{{Select_Month_Lidor.selectedOptionValue}}' = '' 
         OR TO_CHAR(תאריך_יצירה, 'MM/YYYY') = '{{Select_Month_Lidor.selectedOptionValue}}')
)
ORDER BY תאריך_יצירה DESC
LIMIT 200;
