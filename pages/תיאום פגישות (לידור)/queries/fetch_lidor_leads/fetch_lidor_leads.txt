SELECT * FROM lidor_leads_view
WHERE ('{{Select_Month_Lidor.selectedOptionValue}}' = ''
       OR TO_CHAR(תאריך_עדכון, 'MM/YYYY') = '{{Select_Month_Lidor.selectedOptionValue}}')
  AND ('{{Input_Search_Lidor.text}}' = ''
       OR שם_לקוח ILIKE '%' || '{{Input_Search_Lidor.text}}' || '%'
       OR תעודת_זהות ILIKE '%' || '{{Input_Search_Lidor.text}}' || '%')
ORDER BY תאריך_יצירה DESC
LIMIT 200
