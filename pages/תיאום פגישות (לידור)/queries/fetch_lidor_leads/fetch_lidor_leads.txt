SELECT * FROM lidor_leads_view
WHERE (
    '{{Input_Search_Lidor.text}}' != ''
    AND (customer_name ILIKE '%' || '{{Input_Search_Lidor.text}}' || '%'
         OR id_number ILIKE '%' || '{{Input_Search_Lidor.text}}' || '%')
)
OR (
    '{{Input_Search_Lidor.text}}' = ''
    AND touched_by_lidor = TRUE
    AND ('{{Select_Month_Lidor.selectedOptionValue}}' = ''
         OR TO_CHAR(update_date, 'MM/YYYY') = '{{Select_Month_Lidor.selectedOptionValue}}')
    AND ('{{Select_Source.selectedOptionValue}}' = ''
         OR source = '{{Select_Source.selectedOptionValue}}')
    AND ('{{Select_Agent.selectedOptionValue}}' = ''
         OR agent = '{{Select_Agent.selectedOptionValue}}')
)
ORDER BY update_date DESC NULLS LAST
LIMIT 200;
