SELECT "חברת_ביטוח", COUNT(*) AS "כמות_לידים" FROM adi_leads_view WHERE "סוכן_מטפל" IS NOT DISTINCT FROM COALESCE(NULLIF({{AgentFilter.selectedOptionValue}}, ''), "סוכן_מטפל") AND EXTRACT(MONTH FROM "תאריך_עדכון")::text IS NOT DISTINCT FROM COALESCE(NULLIF({{MonthFilter.selectedOptionValue}}, ''), EXTRACT(MONTH FROM "תאריך_עדכון")::text) AND EXTRACT(YEAR FROM "תאריך_עדכון")::text IS NOT DISTINCT FROM COALESCE(NULLIF({{YearFilter.selectedOptionValue}}, ''), EXTRACT(YEAR FROM "תאריך_עדכון")::text) GROUP BY "חברת_ביטוח" ORDER BY "כמות_לידים" DESC
