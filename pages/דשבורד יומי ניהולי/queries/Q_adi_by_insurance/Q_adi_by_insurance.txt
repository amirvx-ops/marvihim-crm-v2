SELECT
  COALESCE(insurance_company, 'לא שויכה') AS "חברת_ביטוח",
  COUNT(*) AS "סגירות",
  COALESCE(SUM(closing_amount), 0) AS "סכום_סגירה",
  COALESCE(SUM(agent_appointment_amount), 0) AS "סכום_מינוי"
FROM leads
WHERE user_updated_at = '{{DatePicker_Date.selectedDate}}'::date
  AND touched_by_adi = TRUE
  AND insurance_company IS NOT NULL
  AND insurance_company <> ''
GROUP BY insurance_company
ORDER BY "סגירות" DESC;
