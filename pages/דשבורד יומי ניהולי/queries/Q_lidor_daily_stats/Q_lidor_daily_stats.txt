SELECT
  (SELECT COUNT(*) FROM lidor_leads_view
   WHERE masleka_date = '{{DatePicker_Date.selectedDate}}'::date
     AND touched_by_lidor = TRUE) AS "מסלקות_הוזמנו",

  (SELECT COUNT(*) FROM lidor_leads_view
   WHERE lidor_status = 'נקבעה פגישה'
     AND update_date = '{{DatePicker_Date.selectedDate}}'::date
     AND touched_by_lidor = TRUE) AS "פגישות_נקבעו",

  (SELECT COUNT(*) FROM lidor_leads_view
   WHERE close_date = '{{DatePicker_Date.selectedDate}}'::date
     AND touched_by_lidor = TRUE) AS "סגירות",

  (SELECT COALESCE(SUM(closing_amount), 0) FROM lidor_leads_view
   WHERE close_date = '{{DatePicker_Date.selectedDate}}'::date
     AND touched_by_lidor = TRUE) AS "סכום_סגירות",

  (SELECT
     CASE WHEN COUNT(meeting_held) = 0 THEN 0
          ELSE ROUND(100.0 * COUNT(*) FILTER (WHERE meeting_held = 'כן')
                      / COUNT(meeting_held), 1)
     END
   FROM lidor_leads_view
   WHERE touched_by_lidor = TRUE
     AND DATE_TRUNC('month', meeting_date)
         = DATE_TRUNC('month', '{{DatePicker_Date.selectedDate}}'::timestamp)
     AND meeting_held IS NOT NULL) AS "אחוז_פגישות_מתקיימות_החודש";
