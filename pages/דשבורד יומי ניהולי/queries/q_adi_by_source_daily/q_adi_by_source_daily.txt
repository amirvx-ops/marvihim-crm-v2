SELECT COALESCE(p.name, 'לא ידוע') AS "מקור", COUNT(*) AS "כמות" FROM leads l LEFT JOIN partners p ON l.partner_id = p.id WHERE l.type = 'בדיקת תיק מלאה' AND ( l.main_status_date_of_update BETWEEN '{{moment(_DatePicker_Start.selectedDate).format("YYYY-MM-DD")}}'::date AND '{{moment(DatePicker_End.selectedDate).format("YYYY-MM-DD")}}'::date OR l.user_updated_at BETWEEN '{{moment(_DatePicker_Start.selectedDate).format("YYYY-MM-DD")}}'::date AND '{{moment(DatePicker_End.selectedDate).format("YYYY-MM-DD")}}'::date ) AND ( ({{AgentSelect.selectedOptionValue}}::int = 1 AND l.touched_by_adi = TRUE) OR ({{AgentSelect.selectedOptionValue}}::int = 2 AND l.touched_by_yuli = TRUE) OR ({{AgentSelect.selectedOptionValue}}::int = 3 AND (l.touched_by_adi = TRUE OR l.touched_by_yuli = TRUE)) ) GROUP BY p.name ORDER BY "כמות" DESC
